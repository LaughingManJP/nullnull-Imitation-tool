M5Stack StopWatch · スケッチ

NullLife

丸い画面全体が液体金属のような鏡面の膜になり、その下で「何か」がゆっくり動き、呼吸し、鼓動します。構うほど呼吸と鼓動が速くなり、放っておくと落ち着いていきます。

  • 触る画面をクリック — ビクッと縮んで逃げる（大きな波紋）
  • なぞる押したままドラッグ — 指の跡に波が立ち、中の生き物が寄ってくる
  • ボタン A黄色いボタン か A — 映り込む景色：銀 / 夕暮れ / ネオン / 金
  • ボタン B青いボタン か B — 明るさ 4 段階

NullLife.ino を書き換えずに、M5Unified 互換の描画レイヤーと一緒に WebAssembly にコンパイルしてブラウザで動かしています。実機ではコア0で計算、コア1で描画しますが、ここでは同じ処理を1つのスレッドで順番に実行します。速度と色味は実機と少し異なります。

スケッチのソース（NullLife.ino） 
// NullLife  -  null²風「生きている鏡面」 for M5Stack StopWatch
//
// 丸い画面全体が液体金属のような鏡面の膜になり、
// 膜の下で「何か」がゆっくり動き、呼吸し、鼓動します。
//   ・触る      : ビクッと縮んで逃げる(大きな波紋)
//   ・なぞる    : 指の跡に波が立ち、中の生き物が興味を持って寄ってくる
//   ・構うほど  : 呼吸と鼓動が速くなり、放っておくと落ち着いていく
//   ・ボタンA   : 映り込む景色(配色)を切り替え  銀 / 夕暮れ / ネオン / 金
//   ・ボタンB   : 明るさ切り替え
//
// 必要なもの:
//   ボードマネージャー M5Stack >= 3.3.7 / ボード: M5StopWatch (PSRAM有効)
//   ライブラリ M5Unified >= 0.2.15, M5GFX >= 0.2.21
//
// しくみ:
//   234x234 の格子で「高さ」を計算 → 傾きから鏡に映る方向を求め、
//   景色テクスチャを引いて 2倍に拡大して 466x466 に描画。
//   計算はコア0、描画はコア1で並行して行います。

#include <M5Unified.h>
#include <math.h>

static constexpr int   W      = 466;          // 画面サイズ
static constexpr int   CELLS  = 233;          // 格子のマス数 (x2 = 466)
static constexpr int   G      = CELLS + 1;    // 格子点の数
static constexpr float C      = (G - 1) * 0.5f;
static constexpr float R      = (G - 1) * 0.5f;
static constexpr int   ENV    = 128;          // 景色テクスチャのサイズ
static constexpr int   NLUMP  = 6;            // 膜の下の「生き物」の数

struct UV { uint16_t u, v; };                 // 8.8 固定小数点

static uint16_t* env;          // 景色テクスチャ (内部RAM)
static uint16_t* strip[2];     // 描画用バッファ (DMA)
static UV*       uvBuf[2];     // 反射方向 (PSRAM, ダブルバッファ)
static float*    hField;       // 高さ
static int16_t*  waveA;        // 波紋 (現在)
static int16_t*  waveB;        // 波紋 (1つ前)
static uint8_t*  mask;         // 円の内側=1
static float*    dome;         // 中心からの距離^2 (0..1)
static float     expLut[256];

struct Lump { float x, y, vx, vy, baseAmp, sig, phase, theta; };
static Lump lumps[NLUMP];

struct Input { bool touching; bool tapped; float tx, ty; float dt; };
static Input simIn;

static SemaphoreHandle_t semGo, semDone;
static int front = 0;

// 生き物の状態
static float simT = 0, activity = 0, fear = 0, wake = 0;
static float breathPhase = 0, hbPhase = 0, twitchTimer = 3;

static int theme = 0;
static const uint8_t brightSteps[] = {80, 150, 220, 255};
static int brightIdx = 2;

// ---------------------------------------------------------------- 景色
struct Col { float r, g, b; };
struct Theme { Col skyTop, horizon, ground, band, sun, glow; };
static const Theme THEMES[] = {
  // 銀 (null²っぽい昼の鏡面)
  {{0.55f,0.72f,0.95f},{0.95f,0.97f,1.00f},{0.07f,0.08f,0.10f},{0.38f,0.40f,0.44f},{1.0f,1.0f,1.0f},{0.85f,0.92f,1.0f}},
  // 夕暮れ
  {{0.16f,0.10f,0.38f},{1.00f,0.55f,0.30f},{0.05f,0.03f,0.08f},{0.42f,0.18f,0.26f},{1.0f,0.85f,0.5f},{1.0f,0.55f,0.30f}},
  // ネオン
  {{0.01f,0.01f,0.04f},{0.12f,0.05f,0.28f},{0.00f,0.00f,0.02f},{0.00f,0.75f,0.90f},{1.0f,0.2f,0.8f},{0.30f,0.90f,1.0f}},
  // 金
  {{0.95f,0.82f,0.50f},{1.00f,0.97f,0.82f},{0.16f,0.09f,0.02f},{0.62f,0.42f,0.12f},{1.0f,1.0f,0.9f},{1.0f,0.90f,0.60f}},
};
static constexpr int NTHEME = sizeof(THEMES) / sizeof(THEMES[0]);

static inline float clamp01(float v) { return v < 0 ? 0 : (v > 1 ? 1 : v); }
static inline Col lerpC(Col a, Col b, float t) {
  return {a.r + (b.r - a.r) * t, a.g + (b.g - a.g) * t, a.b + (b.b - a.b) * t};
}

static void buildEnv(int th) {
  const Theme& T = THEMES[th];
  for (int j = 0; j < ENV; ++j) {
    for (int i = 0; i < ENV; ++i) {
      float x = (i - 63.5f) / 64.0f;
      float y = (j - 63.5f) / 64.0f;               // 下向きが正
      float hz = 0.08f + 0.06f * sinf(x * 2.2f);   // ゆるく波打つ地平線
      Col c;
      if (y < hz) {                                // 空
        float s = clamp01((hz - y) / (hz + 1.0f));
        c = lerpC(T.horizon, T.skyTop, powf(s, 0.6f));
        float dx = x - 0.35f, dy = y + 0.55f;
        float sun = expf(-(dx * dx + dy * dy) / 0.02f);
        c.r += T.sun.r * sun; c.g += T.sun.g * sun; c.b += T.sun.b * sun;
      } else {                                     // 地面と映り込む構造物
        float gnd = clamp01((y - hz) / (1.0f - hz));
        float bands = 0.5f + 0.5f * sinf(x * 9.0f + sinf(x * 3.0f) * 2.0f);
        bands = powf(bands, 6.0f) * (1.0f - gnd);
        c = lerpC(T.ground, T.band, bands);
      }
      float g = expf(-(y - hz) * (y - hz) / 0.0015f);  // 地平線の光
      c.r += T.glow.r * g * 0.8f; c.g += T.glow.g * g * 0.8f; c.b += T.glow.b * g * 0.8f;
      uint16_t col = M5.Display.color565(clamp01(c.r) * 255, clamp01(c.g) * 255, clamp01(c.b) * 255);
      env[j * ENV + i] = __builtin_bswap16(col);   // 色がおかしい場合はこの bswap を外す
    }
  }
}

// ---------------------------------------------------------------- シミュレーション
static inline float frand() { return (esp_random() & 0xFFFF) / 65535.0f; }

static void inject(float gx, float gy, int rad, int power) {
  int cx = (int)gx, cy = (int)gy;
  for (int y = cy - rad; y <= cy + rad; ++y) {
    if (y < 1 || y >= G - 1) continue;
    for (int x = cx - rad; x <= cx + rad; ++x) {
      if (x < 1 || x >= G - 1) continue;
      int d2 = (x - cx) * (x - cx) + (y - cy) * (y - cy);
      if (d2 > rad * rad) continue;
      int i = y * G + x;
      if (!mask[i]) continue;
      int v = waveA[i] + power * (rad * rad - d2) / (rad * rad);
      waveA[i] = v > 30000 ? 30000 : (v < -30000 ? -30000 : v);
    }
  }
}

static inline float pulse(float x) { return expf(-(x * x) / (2 * 0.025f * 0.025f)); }

static void simulate(const Input& in, UV* out) {
  float dt = in.dt > 0.05f ? 0.05f : in.dt;
  simT += dt;

  // 気分: 触られると興奮し、放っておくと落ち着く
  if (in.touching) activity = fminf(1.0f, activity + dt * 1.5f);
  else             activity = fmaxf(0.0f, activity - dt / 20.0f);
  fear *= expf(-dt * 1.2f);
  wake  = fminf(1.0f, wake + dt / 3.0f);             // 起動時にゆっくり目覚める

  if (in.tapped) {                                   // ビクッ
    fear = 1.0f;
    inject(in.tx, in.ty, 6, 6000);
    for (auto& L : lumps) {
      float dx = L.x - in.tx, dy = L.y - in.ty;
      float d = fmaxf(sqrtf(dx * dx + dy * dy), 1.0f);
      float k = 120.0f / (1.0f + d / 40.0f);
      L.vx += dx / d * k; L.vy += dy / d * k;
    }
  } else if (in.touching) {
    inject(in.tx, in.ty, 3, (int)(700 * (0.5f + activity)));
  }

  // ときどき勝手にピクッと動く
  twitchTimer -= dt;
  if (twitchTimer < 0) {
    float a = frand() * 6.2832f, r = sqrtf(frand()) * 0.6f * R;
    inject(C + cosf(a) * r, C + sinf(a) * r, 4, 1200 + (int)(frand() * 1500));
    twitchTimer = (2.0f + frand() * 5.0f) * (1.0f - 0.5f * activity);
  }

  // 呼吸と鼓動
  float period = fmaxf(1.6f, 4.5f - 1.8f * activity - 1.2f * fear);
  breathPhase += dt * 6.2832f / period;
  float breath = sinf(breathPhase);
  hbPhase += dt / (1.3f - 0.5f * activity - 0.4f * fear);
  if (hbPhase >= 1.0f) hbPhase -= 1.0f;
  float hb = pulse(hbPhase - 0.05f) + 0.6f * pulse(hbPhase - 0.21f);

  // 膜の下の生き物の動き
  float speed = 0.6f + 1.2f * activity + 1.5f * fear;
  for (int n = 0; n < NLUMP; ++n) {
    Lump& L = lumps[n];
    L.theta += dt * speed * (0.8f * sinf(simT * 0.37f + L.phase) + 0.6f * sinf(simT * 0.83f + L.phase * 1.7f));
    float fx = cosf(L.theta) * 18.0f * speed;
    float fy = sinf(L.theta) * 18.0f * speed;

    float dx = L.x - C, dy = L.y - C, r = sqrtf(dx * dx + dy * dy) + 0.001f;
    float lim = 0.58f * R;
    if (r > lim) { fx -= dx / r * (r - lim) * 3.0f; fy -= dy / r * (r - lim) * 3.0f; }

    for (int m = 0; m < NLUMP; ++m) {                // 互いに少し離れる
      if (m == n) continue;
      float ox = L.x - lumps[m].x, oy = L.y - lumps[m].y;
      float d = sqrtf(ox * ox + oy * oy) + 0.001f;
      float want = L.sig + lumps[m].sig;
      if (d < want) { fx += ox / d * (want - d) * 2.0f; fy += oy / d * (want - d) * 2.0f; }
    }

    if (in.touching) {                               // 指に興味を持つ(近すぎると離れる)
      float tx = in.tx - L.x, ty = in.ty - L.y;
      float d = sqrtf(tx * tx + ty * ty) + 0.001f;
      if (d > 30) { fx += tx / d * 25.0f * activity; fy += ty / d * 25.0f * activity; }
      else        { fx -= tx / d * 40.0f * (30 - d) / 30; fy -= ty / d * 40.0f * (30 - d) / 30; }
    }

    L.vx = (L.vx + fx * dt) * (1.0f - 1.2f * dt);
    L.vy = (L.vy + fy * dt) * (1.0f - 1.2f * dt);
    L.x += L.vx * dt; L.y += L.vy * dt;
  }

  // 波紋の伝わり
  for (int y = 1; y < G - 1; ++y) {
    int i = y * G + 1;
    for (int x = 1; x < G - 1; ++x, ++i) {
      if (!mask[i]) { waveB[i] = 0; continue; }
      int v = ((waveA[i - 1] + waveA[i + 1] + waveA[i - G] + waveA[i + G]) >> 1) - waveB[i];
      v -= v >> 5;
      waveB[i] = v > 30000 ? 30000 : (v < -30000 ? -30000 : v);
    }
  }
  int16_t* tmp = waveA; waveA = waveB; waveB = tmp;

  // 高さ = ふくらんだ膜 + 波紋 + 生き物
  float domeDepth = 22.0f + 3.0f * breath + 4.0f * hb;
  const float waveScale = 1.0f / 200.0f;
  for (int i = 0; i < G * G; ++i) hField[i] = -domeDepth * dome[i] + waveA[i] * waveScale;

  float gain = wake * (1.0f + 0.12f * breath + 0.08f * hb) * (1.0f - 0.45f * fear);
  for (auto& L : lumps) {
    float amp = L.baseAmp * gain * (0.85f + 0.15f * sinf(simT * 1.3f + L.phase));
    float sig = L.sig * (1.0f + 0.08f * breath);
    float inv = 32.0f / (sig * sig);
    float rad = sig * 2.8f;
    int x0 = max(0, (int)(L.x - rad)), x1 = min(G - 1, (int)(L.x + rad));
    int y0 = max(0, (int)(L.y - rad)), y1 = min(G - 1, (int)(L.y + rad));
    for (int y = y0; y <= y1; ++y) {
      float dy2 = (y - L.y) * (y - L.y);
      float* row = hField + y * G;
      for (int x = x0; x <= x1; ++x) {
        float dx = x - L.x;
        int idx = (int)((dx * dx + dy2) * inv);
        if (idx < 256) row[x] += amp * expLut[idx];
      }
    }
  }

  // 傾き → 鏡に映る方向
  const float k = 38.0f * 256.0f;
  const float maxUV = (ENV - 1) * 256.0f + 255.0f;
  for (int y = 0; y < G; ++y) {
    int ym = y > 0 ? y - 1 : 0, yp = y < G - 1 ? y + 1 : G - 1;
    for (int x = 0; x < G; ++x) {
      int xm = x > 0 ? x - 1 : 0, xp = x < G - 1 ? x + 1 : G - 1;
      float gx = hField[y * G + xp] - hField[y * G + xm];
      float gy = hField[yp * G + x] - hField[ym * G + x];
      float u = 64.0f * 256.0f + gx * k;
      float v = 64.0f * 256.0f + gy * k;
      u = u < 0 ? 0 : (u > maxUV ? maxUV : u);
      v = v < 0 ? 0 : (v > maxUV ? maxUV : v);
      out[y * G + x] = {(uint16_t)u, (uint16_t)v};
    }
  }
}

static void simTask(void*) {
  for (;;) {
    xSemaphoreTake(semGo, portMAX_DELAY);
    simulate(simIn, uvBuf[front ^ 1]);
    xSemaphoreGive(semDone);
  }
}

// ---------------------------------------------------------------- 描画
static inline uint16_t envAt(uint32_t u, uint32_t v) { return env[(v >> 8) * ENV + (u >> 8)]; }

static void renderFrame(const UV* uv) {
  M5.Display.startWrite();
  int s = 0;
  for (int gy = 0; gy < CELLS; ++gy) {
    const UV* r0 = uv + gy * G;
    const UV* r1 = r0 + G;
    uint16_t* row0 = strip[s];
    uint16_t* row1 = row0 + W;
    for (int cx = 0; cx < CELLS; ++cx) {
      const UV a = r0[cx], b = r0[cx + 1], c = r1[cx], d = r1[cx + 1];
      row0[cx * 2]     = envAt(a.u, a.v);
      row0[cx * 2 + 1] = envAt((a.u + b.u) >> 1, (a.v + b.v) >> 1);
      row1[cx * 2]     = envAt((a.u + c.u) >> 1, (a.v + c.v) >> 1);
      row1[cx * 2 + 1] = envAt((a.u + b.u + c.u + d.u) >> 2, (a.v + b.v + c.v + d.v) >> 2);
    }
    M5.Display.waitDMA();
    M5.Display.pushImageDMA(0, gy * 2, W, 2, strip[s]);
    s ^= 1;
  }
  M5.Display.waitDMA();
  M5.Display.endWrite();
}

// ---------------------------------------------------------------- setup / loop
static void fail(const char* msg) {
  M5.Display.setTextDatum(middle_center);
  M5.Display.drawString(msg, W / 2, W / 2);
  Serial.println(msg);
  while (true) delay(1000);
}

void setup() {
  auto cfg = M5.config();
  M5.begin(cfg);
  Serial.begin(115200);
  M5.Display.fillScreen(TFT_BLACK);
  M5.Display.setBrightness(brightSteps[brightIdx]);

  env      = (uint16_t*)heap_caps_malloc(ENV * ENV * 2, MALLOC_CAP_INTERNAL | MALLOC_CAP_8BIT);
  strip[0] = (uint16_t*)heap_caps_malloc(W * 2 * 2, MALLOC_CAP_DMA);
  strip[1] = (uint16_t*)heap_caps_malloc(W * 2 * 2, MALLOC_CAP_DMA);
  uvBuf[0] = (UV*)ps_malloc(G * G * sizeof(UV));
  uvBuf[1] = (UV*)ps_malloc(G * G * sizeof(UV));
  hField   = (float*)ps_malloc(G * G * sizeof(float));
  dome     = (float*)ps_malloc(G * G * sizeof(float));
  waveA    = (int16_t*)ps_calloc(G * G, sizeof(int16_t));
  waveB    = (int16_t*)ps_calloc(G * G, sizeof(int16_t));
  mask     = (uint8_t*)ps_malloc(G * G);
  if (!env || !strip[0] || !strip[1] || !uvBuf[0] || !uvBuf[1] ||
      !hField || !dome || !waveA || !waveB || !mask) {
    fail("Memory error (PSRAM?)");
  }

  for (int y = 0; y < G; ++y)
    for (int x = 0; x < G; ++x) {
      float dx = (x - C) / R, dy = (y - C) / R, r2 = dx * dx + dy * dy;
      dome[y * G + x] = r2;
      mask[y * G + x] = r2 < 0.98f;
    }
  for (int i = 0; i < 256; ++i) expLut[i] = expf(-i / 32.0f);

  for (auto& L : lumps) {
    float a = frand() * 6.2832f, r = sqrtf(frand()) * 0.45f * R;
    L.x = C + cosf(a) * r; L.y = C + sinf(a) * r;
    L.vx = L.vy = 0;
    L.baseAmp = 9.0f + frand() * 7.0f;
    L.sig     = 14.0f + frand() * 12.0f;
    L.phase   = frand() * 6.2832f;
    L.theta   = frand() * 6.2832f;
  }

  buildEnv(theme);

  // 最初の1枚
  simIn = {false, false, 0, 0, 0.016f};
  simulate(simIn, uvBuf[front]);

  semGo   = xSemaphoreCreateBinary();
  semDone = xSemaphoreCreateBinary();
  xTaskCreatePinnedToCore(simTask, "sim", 8192, nullptr, 1, nullptr, 0);
}

void loop() {
  static uint32_t last = millis();
  static uint32_t fpsTimer = millis();
  static int frames = 0;

  M5.update();
  if (M5.BtnA.wasPressed()) { theme = (theme + 1) % NTHEME; buildEnv(theme); }
  if (M5.BtnB.wasPressed()) {
    brightIdx = (brightIdx + 1) % sizeof(brightSteps);
    M5.Display.setBrightness(brightSteps[brightIdx]);
  }

  auto t = M5.Touch.getDetail();
  uint32_t now = millis();
  simIn.touching = t.isPressed();
  simIn.tapped   = t.wasPressed();
  simIn.tx       = t.x * 0.5f;
  simIn.ty       = t.y * 0.5f;
  simIn.dt       = (now - last) / 1000.0f;
  last = now;

  xSemaphoreGive(semGo);              // 次のフレームを裏で計算
  renderFrame(uvBuf[front]);          // 今のフレームを描画
  xSemaphoreTake(semDone, portMAX_DELAY);
  front ^= 1;

  if (++frames, millis() - fpsTimer >= 3000) {
    Serial.printf("fps: %.1f\n", frames * 1000.0f / (millis() - fpsTimer));
    frames = 0; fpsTimer = millis();
  }
}
