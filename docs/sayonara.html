ホモサピエンス

夜のヌルの森。空には星と光のドローン、下半分は null² のような鏡の床で、すべてが映り込んでいます。真ん中のストップウォッチは、数字が変わるたびに光のかけらになってほどけ、風にのって空へ昇っていきます。時間を、少しずつ手放していく時計です。

  • 待つほどけた数字のかけらが空へ昇り、ドローンの光になる。ときどき「∅（ヌル）」や森の形に集まって、またほどける
  • 触る・なぞる鏡の床に光る足跡と波紋が残る。足跡はやがて風に溶けて、空へ舞い上がる
  • ボタン A黄色いボタン か A — さようなら。すべてがほどけて空へ帰り、静かになってから、また 0 から始まる
  • ボタン B青いボタン か B — 明るさ 4 段階

落合陽一さんが null²（大阪・関西万博）のために作った楽曲「さようならホモサピエンス」（作詞：落合陽一+AI、作曲：AI）のイメージから作った、非公式のオリジナルスケッチです。歌詞は使っていません。実機用の SayonaraHomoSapiens.ino を書き換えずに WebAssembly にしてブラウザで動かしています。速度と色味は実機と少し異なります。

スケッチのソース（SayonaraHomoSapiens.ino） 
// SayonaraHomoSapiens  -  「さようならホモサピエンス」をイメージした M5Stack StopWatch スケッチ
//
// 夜のヌルの森。空には星と、光のドローン。下半分は null² のような鏡の床で、
// 空も光もすべてが映り込んでいます。
//   ・時間        : 真ん中のストップウォッチの数字は、変わるたびに光のかけらになって
//                   ほどけ、風にのって空へ昇っていく(時間を手放していく)
//   ・触る/なぞる : 鏡の床に光る足跡が残り、波紋が広がる。足跡はやがて風に溶けて空へ
//   ・ドローン    : 空へ昇ったかけらはドローンの光になり、ときどき
//                   「∅(ヌル)」や「森」の形に集まって、またほどける
//   ・ボタンA     : さようなら。すべてがほどけて空へ帰り、静かになってから、また0から
//   ・ボタンB     : 明るさ切り替え
//
// 必要なもの:
//   ボードマネージャー M5Stack >= 3.3.7 / ボード: M5StopWatch (PSRAM有効)
//   ライブラリ M5Unified >= 0.2.15, M5GFX >= 0.2.21
//
// しくみ:
//   466x466 の画面をまるごと PSRAM 上の配列に描き(空→光→鏡に映す→床)、
//   8行ずつ内部RAMに写して DMA で転送します。1つのコアだけで動きます。

#include <M5Unified.h>
#include <math.h>

static constexpr int   W  = 466;
static constexpr int   H  = 466;
static constexpr float CX = 233.0f;
static constexpr float CY = 233.0f;
static constexpr float RAD = 233.0f;
static constexpr int   STRIP_ROWS = 8;

static constexpr int MAX_PARTS  = 900;   // 光のかけら
static constexpr int MAX_DRONES = 150;   // 空のドローン
static constexpr int NSTARS     = 150;
static constexpr int MAX_STEPS  = 40;    // 足跡
static constexpr int MAX_RIPPLE = 16;    // 鏡の床の波紋

static uint16_t* fb;                     // 画面 (PSRAM, RGB565)
static uint16_t* strip[2];               // 転送用 (内部RAM, DMA)
static float     glowX[W];               // 地平線の光の横方向の広がり

static const uint8_t brightSteps[] = {80, 150, 220, 255};
static int brightIdx = 2;

// ---------------------------------------------------------------- 色
static inline uint16_t rgb(int r, int g, int b) {
  r = r < 0 ? 0 : (r > 255 ? 255 : r);
  g = g < 0 ? 0 : (g > 255 ? 255 : g);
  b = b < 0 ? 0 : (b > 255 ? 255 : b);
  return (uint16_t)(((r & 0xF8) << 8) | ((g & 0xFC) << 3) | (b >> 3));
}

// 光を足す (r,g,b は 0..255 の明るさ)
static inline void addPix(int x, int y, int r, int g, int b) {
  if ((unsigned)x >= (unsigned)W || (unsigned)y >= (unsigned)H) return;
  uint16_t& c = fb[y * W + x];
  int cr = ((c >> 11) & 31) + (r >> 3);
  int cg = ((c >> 5) & 63) + (g >> 2);
  int cb = (c & 31) + (b >> 3);
  if (cr > 31) cr = 31;
  if (cg > 63) cg = 63;
  if (cb > 31) cb = 31;
  c = (uint16_t)((cr << 11) | (cg << 5) | cb);
}

// ぼんやり光る点
static void glow(float fx, float fy, float radius, float r, float g, float b) {
  int x0 = (int)floorf(fx - radius), x1 = (int)ceilf(fx + radius);
  int y0 = (int)floorf(fy - radius), y1 = (int)ceilf(fy + radius);
  float inv = 1.0f / ((radius + 0.5f) * (radius + 0.5f));
  for (int y = y0; y <= y1; ++y)
    for (int x = x0; x <= x1; ++x) {
      float dx = x - fx, dy = y - fy;
      float w = 1.0f - (dx * dx + dy * dy) * inv;
      if (w <= 0) continue;
      w *= w;
      addPix(x, y, (int)(r * w), (int)(g * w), (int)(b * w));
    }
}

static inline float frand() { return (esp_random() & 0xFFFF) / 65535.0f; }
static inline float insideCircle(float x, float y) {
  float dx = x - CX, dy = y - CY;
  return dx * dx + dy * dy < RAD * RAD;
}

// ---------------------------------------------------------------- 世界の状態
static float simT = 0;
static float horizon = 292;              // 空と鏡の床のさかい目
static float light = 0;                  // 世界の明るさ (さようなら で落ちる)
static float wind = 0;

enum Mode { LIVING, GOODBYE };
static Mode  mode = LIVING;
static float modeT = 0;
static uint32_t clockStart = 0;

// 光のかけら
struct Part { float x, y, vx, vy, life, maxLife, r, g, b; bool on; };
static Part parts[MAX_PARTS];
static int  partNext = 0;

static void spawn(float x, float y, float vx, float vy, float life, float r, float g, float b) {
  for (int k = 0; k < MAX_PARTS; ++k) {
    Part& p = parts[(partNext + k) % MAX_PARTS];
    if (p.on) continue;
    p = {x, y, vx, vy, life, life, r, g, b, true};
    partNext = (partNext + k + 1) % MAX_PARTS;
    return;
  }
}

// ドローン
struct Drone { float x, y, vx, vy, tx, ty, phase, hue; bool on, target; };
static Drone drones[MAX_DRONES];
static int   droneCount = 0;
enum Shape { SHAPE_NONE, SHAPE_NULL, SHAPE_FOREST };
static Shape shape = SHAPE_NONE;
static float shapeTimer = 9;

// 星
struct Star { int16_t x, y; float phase, speed, level; };
static Star stars[NSTARS];

// 足跡と波紋
struct Step { float x, y, ang, life; bool left, on; };
static Step steps[MAX_STEPS];
static int  stepNext = 0;
static bool leftFoot = true;
static float lastStepX = -999, lastStepY = -999;

struct Ripple { float x, y, r, life; bool on; };
static Ripple ripples[MAX_RIPPLE];

// ---------------------------------------------------------------- 7セグの時計 (光の点でできている)
static constexpr int   DIGITS = 4;
static const float     DIGIT_X[DIGITS] = {CX - 104, CX - 50, CX + 14, CX + 68};
static constexpr float DIGIT_Y = 168, DIGIT_W = 36, DIGIT_H = 76;
static const uint8_t   SEGMAP[10] = {0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F};
// a b c d e f g の端点 (0..1)
static const float SEG[7][4] = {
  {0, 0, 1, 0}, {1, 0, 1, 0.5f}, {1, 0.5f, 1, 1}, {0, 1, 1, 1},
  {0, 0.5f, 0, 1}, {0, 0, 0, 0.5f}, {0, 0.5f, 1, 0.5f},
};
static float segAlpha[DIGITS][7];
static int   shownDigit[DIGITS] = {-1, -1, -1, -1};

static inline void segPoint(int d, int s, float t, float& x, float& y) {
  float u = SEG[s][0] + (SEG[s][2] - SEG[s][0]) * t;
  float v = SEG[s][1] + (SEG[s][3] - SEG[s][1]) * t;
  x = DIGIT_X[d] + u * DIGIT_W + (0.5f - v) * 9.0f;   // 少し斜体
  y = DIGIT_Y + v * DIGIT_H;
}

// 消えるセグメントを、光のかけらにして空へ放つ
static void dissolveSegment(int d, int s) {
  for (int i = 0; i <= 8; i += 2) {
    float x, y;
    segPoint(d, s, i / 8.0f, x, y);
    spawn(x, y, (frand() - 0.5f) * 14, -8 - frand() * 16, 3.5f + frand() * 2.5f, 170, 205, 255);
  }
}

static void setClock(int value[DIGITS], bool dissolve) {
  for (int d = 0; d < DIGITS; ++d) {
    if (value[d] == shownDigit[d]) continue;
    uint8_t oldMask = shownDigit[d] < 0 ? 0 : SEGMAP[shownDigit[d]];
    uint8_t newMask = value[d] < 0 ? 0 : SEGMAP[value[d]];
    for (int s = 0; s < 7; ++s) {
      bool was = oldMask & (1 << s), now = newMask & (1 << s);
      if (was && !now) {
        if (dissolve && segAlpha[d][s] > 0.3f) dissolveSegment(d, s);
        segAlpha[d][s] = 0;
      }
    }
    shownDigit[d] = value[d];
  }
}

// ---------------------------------------------------------------- 空・鏡
static void drawSky() {
  int hz = (int)horizon;
  for (int y = 0; y < hz && y < H; ++y) {
    float t = (float)y / horizon;                     // 0=天頂, 1=地平線
    float r = 3 + 10 * t * t, g = 4 + 12 * t * t, b = 14 + 40 * t * t;
    float hg = expf(-((horizon - y) / 26.0f) * ((horizon - y) / 26.0f));
    r += 26 * hg; g += 18 * hg; b += 36 * hg;
    uint16_t base = rgb((int)(r * light), (int)(g * light), (int)(b * light));
    uint16_t* row = fb + y * W;
    for (int x = 0; x < W; ++x) row[x] = base;
    // 地平線の真ん中に、森の光がにじむ
    if (horizon - y < 60) {
      float k = expf(-((horizon - y) / 20.0f) * ((horizon - y) / 20.0f)) * light;
      for (int x = 0; x < W; ++x) {
        float gx = glowX[x] * k;
        if (gx > 0.02f) addPix(x, y, (int)(120 * gx), (int)(95 * gx), (int)(140 * gx));
      }
    }
  }
  for (auto& s : stars) {
    if (s.y >= hz - 4) continue;
    float tw = 0.55f + 0.45f * sinf(simT * s.speed + s.phase);
    int v = (int)(s.level * tw * (0.35f + 0.65f * light));
    addPix(s.x, s.y, v, v, v + 20);
  }
}

// 地平線より上を、鏡の床にゆらゆら映す
static void reflect() {
  int hz = (int)horizon;
  for (int y = hz; y < H; ++y) {
    int src = 2 * hz - y - 1;
    if (src < 0) src = 0;
    float depth = (y - horizon) / (H - horizon);
    int off = (int)(sinf(y * 0.11f + simT * 1.4f) * (1.0f + 5.0f * depth) +
                    sinf(y * 0.031f - simT * 0.6f) * 2.0f);
    const uint16_t* s = fb + src * W;
    uint16_t* d = fb + y * W;
    for (int x = 0; x < W; ++x) {
      int sx = x + off;
      sx = sx < 0 ? 0 : (sx >= W ? W - 1 : sx);
      uint16_t c = s[sx];
      c = depth < 0.45f ? (uint16_t)(((c >> 1) & 0x7BEF) + ((c >> 2) & 0x39E7))   // 75%
                        : (uint16_t)((c >> 1) & 0x7BEF);                            // 50%
      d[x] = c;
    }
  }
  // さかい目の細い光
  for (int x = 0; x < W; ++x) {
    float k = (0.25f + 0.75f * glowX[x]) * light;
    addPix(x, hz, (int)(90 * k), (int)(100 * k), (int)(140 * k));
  }
}

// ---------------------------------------------------------------- ドローン
static void makeDrone(float x, float y, float r, float g, float b) {
  if (droneCount >= MAX_DRONES) return;
  Drone& d = drones[droneCount++];
  d = {x, y, 0, 0, x, y, frand() * 6.2832f, frand(), true, false};
  (void)r; (void)g; (void)b;
}

static void assignShape(Shape sh) {
  shape = sh;
  int n = droneCount;
  if (n == 0) return;
  for (int i = 0; i < n; ++i) {
    Drone& d = drones[i];
    d.target = true;
    if (sh == SHAPE_NULL) {                            // ∅ : 輪と、ななめの線
      float cxs = CX, cys = 96, rr = 58;
      int ringN = n * 3 / 4;
      if (i < ringN) {
        float a = 6.2832f * i / ringN - 1.5708f;
        d.tx = cxs + cosf(a) * rr; d.ty = cys + sinf(a) * rr;
      } else {
        float t = (float)(i - ringN) / fmaxf(1, n - ringN - 1);
        d.tx = cxs - 70 + t * 140; d.ty = cys + 70 - t * 140;
      }
    } else if (sh == SHAPE_FOREST) {                   // 森 : 時計の両側、地平線から立つ光の木々
      static const float treeX[6] = {-190, -156, -122, 122, 156, 190};
      static const float treeH[6] = {80, 135, 105, 110, 140, 85};
      int tree = i % 6, k = i / 6, per = (n + 5) / 6;
      float t = (float)k / fmaxf(1, per - 1);
      float y = horizon - 6 - t * treeH[tree];
      float spread = sinf(t * 3.1416f) * 11.0f * ((k & 1) ? 1 : -1);   // 枝のふくらみ
      d.tx = CX + treeX[tree] + spread * (t > 0.35f ? 1.0f : 0.2f);
      d.ty = y;
    } else {
      d.target = false;
    }
  }
}

static void updateDrones(float dt) {
  shapeTimer -= dt;
  if (shapeTimer < 0 && mode == LIVING) {
    static int nextShape = 0;
    if (shape == SHAPE_NONE && droneCount > 30) {
      assignShape(nextShape == 0 ? SHAPE_NULL : SHAPE_FOREST);
      nextShape ^= 1;
      shapeTimer = 9;
    } else {
      assignShape(SHAPE_NONE);
      shapeTimer = shape == SHAPE_NONE && droneCount <= 30 ? 3 : 10 + frand() * 4;
    }
  }
  for (int i = 0; i < droneCount; ++i) {
    Drone& d = drones[i];
    float ax, ay;
    if (d.target) {                                    // 形をつくる
      ax = (d.tx - d.x) * 2.2f - d.vx * 2.0f;
      ay = (d.ty - d.y) * 2.2f - d.vy * 2.0f;
    } else {                                           // ゆっくり漂う
      d.phase += dt * (0.3f + 0.2f * d.hue);
      ax = cosf(d.phase) * 6 + wind * 0.15f - d.vx * 0.6f;
      ay = sinf(d.phase * 1.3f) * 4 - d.vy * 0.6f;
      float dx = d.x - CX, dy = d.y - CY, r = sqrtf(dx * dx + dy * dy);
      if (r > RAD - 26) { ax -= dx / r * 20; ay -= dy / r * 20; }
      if (d.y > horizon - 70) ay -= 12;
      if (d.y < 18) ay += 10;
    }
    if (mode == GOODBYE) { ay -= 30; ax *= 0.3f; }
    d.vx += ax * dt; d.vy += ay * dt;
    d.x += d.vx * dt; d.y += d.vy * dt;
  }
  if (mode == GOODBYE) {                               // 空の外へ帰っていく
    int w = 0;
    for (int i = 0; i < droneCount; ++i)
      if (drones[i].y > -10) drones[w++] = drones[i];
    droneCount = w;
  }
}

static void drawDrones() {
  for (int i = 0; i < droneCount; ++i) {
    Drone& d = drones[i];
    if (d.y >= horizon - 1) continue;
    float tw = 0.7f + 0.3f * sinf(simT * 3.0f + d.phase * 5.0f);
    float r, g, b;
    if (d.hue < 0.7f)      { r = 150; g = 210; b = 255; }   // 白っぽい青
    else if (d.hue < 0.88f){ r = 255; g = 160; b = 230; }   // 桃
    else                   { r = 255; g = 235; b = 190; }   // 暖かい白
    float k = d.target ? 1.35f : 1.0f;                  // 形になっているときは少し明るく
    glow(d.x, d.y, d.target ? 2.2f : 1.8f, r * tw * k, g * tw * k, b * tw * k);
  }
}

// ---------------------------------------------------------------- かけら
static void updateParts(float dt) {
  for (auto& p : parts) {
    if (!p.on) continue;
    p.life -= dt;
    p.vy -= 10.0f * dt;                                // ふわっと昇る
    p.vx += (wind + sinf(p.y * 0.03f + simT) * 8.0f) * dt;
    p.vx *= 1.0f - 0.6f * dt;
    p.vy *= 1.0f - 0.25f * dt;
    p.x += p.vx * dt; p.y += p.vy * dt;
    bool high = p.y < 70 + 60 * frand();
    if (p.life <= 0 || high || !insideCircle(p.x, p.y)) {
      if (high && mode == LIVING && insideCircle(p.x, p.y) && frand() < 0.6f)
        makeDrone(p.x, p.y, p.r, p.g, p.b);            // 空で、ドローンの光になる
      p.on = false;
    }
  }
}

static void drawParts(bool sky) {
  for (auto& p : parts) {
    if (!p.on) continue;
    if ((p.y < horizon) != sky) continue;
    float a = fminf(1.0f, p.life / p.maxLife * 1.6f) * (0.4f + 0.6f * light);
    glow(p.x, p.y, 1.3f, p.r * a, p.g * a, p.b * a);
  }
}

// ---------------------------------------------------------------- 足跡と波紋
static void putStep(float x, float y, float ang) {
  Step& s = steps[stepNext];
  if (s.on) {                                          // 古い足跡は風に溶かす
    for (int k = 0; k < 6; ++k)
      spawn(s.x + (frand() - 0.5f) * 8, s.y + (frand() - 0.5f) * 12,
            (frand() - 0.5f) * 10, -10 - frand() * 12, 3 + frand() * 2, 255, 205, 160);
  }
  float side = leftFoot ? -1.0f : 1.0f;
  s = {x + cosf(ang) * 7 * side, y + sinf(ang) * 7 * side, ang, 3.0f, leftFoot, true};
  leftFoot = !leftFoot;
  stepNext = (stepNext + 1) % MAX_STEPS;
  for (auto& r : ripples)
    if (!r.on) { r = {x, y, 4, 1.6f, true}; break; }
}

static void updateSteps(float dt) {
  for (auto& s : steps) {
    if (!s.on) continue;
    s.life -= dt;
    if (s.life < 1.0f && frand() < dt * 10)              // 端から少しずつ崩れる
      spawn(s.x + (frand() - 0.5f) * 8, s.y + (frand() - 0.5f) * 12,
            (frand() - 0.5f) * 8, -8 - frand() * 12, 2.5f + frand() * 2, 255, 200, 150);
    if (s.life <= 0) {
      for (int k = 0; k < 8; ++k)
        spawn(s.x + (frand() - 0.5f) * 8, s.y + (frand() - 0.5f) * 12,
              (frand() - 0.5f) * 10, -10 - frand() * 14, 3 + frand() * 2.5f, 255, 205, 160);
      s.on = false;
    }
  }
  for (auto& r : ripples) {
    if (!r.on) continue;
    r.life -= dt; r.r += 55 * dt;
    if (r.life <= 0) r.on = false;
  }
}

static void drawSteps() {
  for (auto& s : steps) {
    if (!s.on) continue;
    float a = fminf(1.0f, s.life / 1.2f) * (0.3f + 0.7f * light);
    // 足の向き。床を斜め上から見ているので、奥行き方向(y)は少しつぶす
    float fx = cosf(s.ang), fy = sinf(s.ang) * 0.6f;
    float n = sqrtf(fx * fx + fy * fy); fx /= n; fy /= n;
    float px = -fy, py = fx;
    float mirror = s.left ? -1.0f : 1.0f;
    auto at = [&](float along, float side, float& x, float& y) {
      x = s.x + fx * along + px * side * mirror;
      y = s.y + fy * along + py * side * mirror * 0.6f;
    };
    float x, y;
    at(-7, 0, x, y);   glow(x, y, 3.0f, 170 * a, 130 * a, 95 * a);   // かかと
    at(-1, 0.6f, x, y); glow(x, y, 2.4f, 120 * a, 90 * a, 70 * a);   // 土ふまず
    at(5, 0.8f, x, y); glow(x, y, 3.6f, 180 * a, 140 * a, 100 * a);  // 母趾球
    static const float toe[5][3] = {{10.5f, -1.8f, 1.5f}, {11.0f, 0.4f, 1.2f}, {10.6f, 2.2f, 1.1f},
                                    {9.8f, 3.8f, 1.0f}, {8.8f, 5.0f, 0.9f}};
    for (auto& t : toe) { at(t[0], t[1], x, y); glow(x, y, t[2], 190 * a, 150 * a, 110 * a); }
  }
  for (auto& r : ripples) {
    if (!r.on) continue;
    float a = r.life / 1.6f * light;
    int n = (int)(r.r * 1.2f) + 12;
    for (int k = 0; k < n; ++k) {
      float t = 6.2832f * k / n;
      addPix((int)(r.x + cosf(t) * r.r), (int)(r.y + sinf(t) * r.r * 0.3f),
             (int)(60 * a), (int)(80 * a), (int)(110 * a));
    }
  }
}

// ---------------------------------------------------------------- 時計を描く
static void drawClock(float dt) {
  if (mode == GOODBYE) return;
  uint32_t sec = (millis() - clockStart) / 1000;
  int value[DIGITS] = {(int)(sec / 600 % 10), (int)(sec / 60 % 10), (int)(sec / 10 % 6), (int)(sec % 10)};
  setClock(value, true);
  for (int d = 0; d < DIGITS; ++d) {
    uint8_t mask = SEGMAP[shownDigit[d]];
    for (int s = 0; s < 7; ++s) {
      if (!(mask & (1 << s))) continue;
      float& al = segAlpha[d][s];
      al = fminf(1.0f, al + dt * 1.2f);                // ゆっくり灯る
      float a = al * al * light;
      for (int i = 0; i <= 8; ++i) {
        float x, y;
        segPoint(d, s, i / 8.0f, x, y);
        float fl = 0.85f + 0.15f * sinf(simT * 4 + i + s * 2 + d * 3);
        glow(x, y, 2.4f, 170 * a * fl, 205 * a * fl, 255 * a * fl);
      }
    }
  }
  // 区切りの点 (ゆっくり明滅)
  float blink = (0.35f + 0.65f * (0.5f + 0.5f * cosf(simT * 6.2832f))) * light;
  glow(CX - 18, DIGIT_Y + DIGIT_H * 0.3f, 2.4f, 120 * blink, 160 * blink, 220 * blink);
  glow(CX - 21, DIGIT_Y + DIGIT_H * 0.7f, 2.4f, 120 * blink, 160 * blink, 220 * blink);
}

// ---------------------------------------------------------------- さようなら
static void beginGoodbye() {
  if (mode == GOODBYE) return;
  mode = GOODBYE; modeT = 0;
  for (int d = 0; d < DIGITS; ++d)                     // 時間をぜんぶ手放す
    for (int s = 0; s < 7; ++s)
      if (shownDigit[d] >= 0 && (SEGMAP[shownDigit[d]] & (1 << s))) dissolveSegment(d, s);
  int none[DIGITS] = {-1, -1, -1, -1};
  setClock(none, false);
  for (auto& s : steps) if (s.on) s.life = fminf(s.life, 0.3f);
  assignShape(SHAPE_NONE);
}

static void updateMode(float dt) {
  modeT += dt;
  if (mode == LIVING) {
    light = fminf(1.0f, light + dt / 3.0f);            // 起動・帰ってきたとき、ゆっくり明るく
  } else {
    if (modeT < 4.0f)       light = fmaxf(0.12f, light - dt / 3.0f);
    else if (modeT > 7.0f) { mode = LIVING; modeT = 0; clockStart = millis(); }
  }
}

// ---------------------------------------------------------------- 転送
static void present() {
  M5.Display.startWrite();
  int k = 0;
  for (int y0 = 0; y0 < H; y0 += STRIP_ROWS) {
    int rows = H - y0 < STRIP_ROWS ? H - y0 : STRIP_ROWS;
    const uint16_t* src = fb + y0 * W;
    uint16_t* dst = strip[k];
    for (int i = 0; i < rows * W; ++i) dst[i] = __builtin_bswap16(src[i]);   // 色がおかしい場合はこの bswap を外す
    M5.Display.waitDMA();
    M5.Display.pushImageDMA(0, y0, W, rows, dst);
    k ^= 1;
  }
  M5.Display.waitDMA();
  M5.Display.endWrite();
}

// ---------------------------------------------------------------- setup / loop
void setup() {
  auto cfg = M5.config();
  M5.begin(cfg);
  Serial.begin(115200);
  M5.Display.fillScreen(TFT_BLACK);
  M5.Display.setBrightness(brightSteps[brightIdx]);

  fb       = (uint16_t*)ps_malloc(W * H * sizeof(uint16_t));
  strip[0] = (uint16_t*)heap_caps_malloc(W * STRIP_ROWS * 2, MALLOC_CAP_DMA);
  strip[1] = (uint16_t*)heap_caps_malloc(W * STRIP_ROWS * 2, MALLOC_CAP_DMA);
  if (!fb || !strip[0] || !strip[1]) {
    M5.Display.setTextDatum(middle_center);
    M5.Display.drawString("Memory error (PSRAM?)", W / 2, H / 2);
    while (true) delay(1000);
  }

  for (int x = 0; x < W; ++x) {
    float dx = (x - CX) / 150.0f;
    glowX[x] = expf(-dx * dx);
  }
  for (auto& s : stars) {
    float x, y;
    do { x = frand() * W; y = frand() * 280; } while (!insideCircle(x, y));
    s = {(int16_t)x, (int16_t)y, frand() * 6.2832f, 0.6f + frand() * 2.2f, 50 + frand() * 160};
  }
  for (int i = 0; i < 24; ++i) {                        // 最初から少しだけ、空に光
    float a = frand() * 6.2832f, r = 60 + frand() * 110;
    makeDrone(CX + cosf(a) * r, 110 + sinf(a) * r * 0.5f, 0, 0, 0);
  }
  clockStart = millis();
}

void loop() {
  static uint32_t last = millis();
  static uint32_t fpsTimer = millis();
  static int frames = 0;

  M5.update();
  uint32_t now = millis();
  float dt = (now - last) / 1000.0f;
  if (dt > 0.05f) dt = 0.05f;
  last = now;
  simT += dt;

  if (M5.BtnA.wasPressed()) beginGoodbye();
  if (M5.BtnB.wasPressed()) {
    brightIdx = (brightIdx + 1) % sizeof(brightSteps);
    M5.Display.setBrightness(brightSteps[brightIdx]);
  }

  // 触る・なぞる → 鏡の床に足跡
  auto t = M5.Touch.getDetail();
  if (t.isPressed() && mode == LIVING) {
    float x = t.x, y = t.y;
    if (t.wasPressed()) { lastStepX = x; lastStepY = y; putStep(x, y, -1.5708f); }
    float dx = x - lastStepX, dy = y - lastStepY;
    if (dx * dx + dy * dy > 26 * 26) {
      putStep(x, y, atan2f(dy, dx));
      lastStepX = x; lastStepY = y;
    }
  }

  wind = 10.0f * sinf(simT * 0.23f) + 5.0f * sinf(simT * 0.71f + 1.0f);
  horizon = 292 + 3.0f * sinf(simT * 0.35f);

  updateMode(dt);
  updateParts(dt);
  updateDrones(dt);
  updateSteps(dt);

  drawSky();                 // 空
  drawDrones();              // ドローン
  drawParts(true);           // 空のかけら
  drawClock(dt);             // 時計 (これも鏡に映る)
  reflect();                 // 鏡の床に映す
  drawSteps();               // 床の足跡と波紋
  drawParts(false);          // 床のかけら
  present();

  if (++frames, millis() - fpsTimer >= 3000) {
    Serial.printf("fps: %.1f  drones: %d\n", frames * 1000.0f / (millis() - fpsTimer), droneCount);
    frames = 0; fpsTimer = millis();
  }
}
